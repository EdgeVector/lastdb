#!/usr/bin/env python3
"""Same-fixture, CPU-pinned diagnostic of exact native routing."""
import argparse
import fcntl
import hashlib
import json
import os
from pathlib import Path
import shutil
import sys
import time

sys.dont_write_bytecode = True
HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parent / 'exact-prefix-benchmark-20261023'))
from benchmark import Bench, COUNT, SCHEMAS, file_hash, key, save, stats
from cleanup import cleanup

REVISIONS = {'before': 'bd517062ed3b86aef3e431d1e2cd64fd9dd08aa5',
             'after': '6965ecb93a4007e50130cc479b4a4040ad03c997'}


def fixture_digest(home):
    digest = hashlib.sha256()
    count = size = 0
    for path in sorted(home.rglob('*')):
        if path.is_symlink():
            raise RuntimeError('Fixture contains a symlink')
        if path.is_file():
            relative = str(path.relative_to(home)).encode()
            digest.update(len(relative).to_bytes(8, 'big') + relative)
            digest.update(bytes.fromhex(file_hash(path)))
            count += 1
            size += path.stat().st_size
        elif not path.is_dir():
            raise RuntimeError('Fixture contains a special file')
    return {'sha256': digest.hexdigest(), 'files': count, 'bytes': size}


class ControlledBench(Bench):
    def start(self):
        previous = os.sched_getaffinity(0)
        os.sched_setaffinity(0, {self.args.server_cpu})
        try:
            super().start()
        finally:
            os.sched_setaffinity(0, previous)
        affinities = [sorted(os.sched_getaffinity(int(task.name)))
                      for task in Path(f'/proc/{self.process.pid}/task').iterdir()]
        if any(cpus != [self.args.server_cpu] for cpus in affinities):
            raise RuntimeError('Daemon thread affinity differs')
        self.remember('cpu-affinity', client=sorted(previous), server_threads=affinities)

    def workloads(self):
        points = {}
        for kind in SCHEMAS:
            points[kind] = [({'HashKey': key(n)} if kind == 'hash' else
                {'HashRangeKey': {'hash': 'measure', 'range': key(n)}}, [n], 1)
                for n in range(COUNT)]
            for selector, numbers, limit in points[kind]:
                self.read(kind, selector, numbers, limit)
        samples = {kind: [] for kind in SCHEMAS}
        blocks = []
        for block in range(8):
            order = ('hash', 'range') if block % 2 == 0 else ('range', 'hash')
            for kind in order:
                batch = [self.read(kind, selector, numbers, limit)
                         for selector, numbers, limit in points[kind]]
                samples[kind].extend(batch)
                blocks.append({'block': block, 'kind': kind, **stats(batch)})
        for kind, values in samples.items():
            phase = f'{kind}_fully_warm_point'
            self.phases[phase] = stats(values)
            self.samples.extend({'phase': phase, 'ms': value} for value in values)
        save(self.output / 'point_blocks.json', blocks)
        self.reads('range_prefix10', 'range', [({'HashRangePrefix': {
            'hash': 'measure', 'prefix': 'key-00012'}}, list(range(120, 130)), 16)] * 256)
        self.reads('range_between64', 'range', [({'HashRangeRange': {
            'hash': 'measure', 'start': key(128), 'end': key(192)}}, list(range(128, 192)), 64)] * 128)
        self.writes('range_fully_warm_queued_write32', 'range', 32, 64, 'queued', 9000)
        self.stop()
        self.start()
        for start in range(0, COUNT, 32):
            numbers = list(range(start, start + 32))
            self.read('range', {'HashRangeKeys': [['measure', key(n)] for n in numbers]}, numbers, 32)
        self.flush()


def execute(args, fixture=None, expected=None):
    bench = ControlledBench(args)
    summary = {'complete': False}
    try:
        if fixture is None:
            bench.start()
            bench.install()
            bench.stop()
            summary['fixture_digest'] = fixture_digest(bench.home)
        else:
            shutil.copytree(fixture, bench.home, dirs_exist_ok=True, copy_function=shutil.copy2)
            observed = fixture_digest(bench.home)
            if observed != expected:
                raise RuntimeError('Cloned fixture differs before startup')
            inputs = json.loads((bench.output / 'inputs.json').read_text())
            inputs.update(controlled_driver_sha256=file_hash(Path(__file__)),
                          fixture_digest=observed, client_cpu=args.client_cpu,
                          server_cpu=args.server_cpu, blocks_per_point_kind=8)
            save(bench.output / 'inputs.json', inputs)
            bench.start()
            bench.workloads()
            summary.update(phases=bench.phases, values_and_write_reopen_verified=True)
        summary['complete'] = True
    except BaseException as error:
        summary['error'] = str(error)
        raise
    finally:
        bench.stop()
        save(bench.output / 'samples.json', bench.samples)
        summary.update(source_revision=args.source_revision, config=args.config,
                       all_owned_processes_stopped=bench.process is None)
        save(bench.output / 'summary.json', summary)
    return bench, summary


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('binaries', 'output', 'scratch'):
        parser.add_argument('--' + name, type=Path, required=True)
    parser.add_argument('--rounds', type=int, default=4)
    parser.add_argument('--client-cpu', type=int, required=True)
    parser.add_argument('--server-cpu', type=int, required=True)
    opts = parser.parse_args()
    allowed = os.sched_getaffinity(0)
    if opts.client_cpu == opts.server_cpu or not {opts.client_cpu, opts.server_cpu} <= allowed:
        raise RuntimeError('Need two distinct allowed CPUs')
    os.sched_setaffinity(0, {opts.client_cpu})
    opts.output.mkdir(parents=True, exist_ok=False)
    opts.scratch.mkdir(mode=0o700, parents=True, exist_ok=False)
    reference = json.loads((HERE / 'provenance.json').read_text())['binaries']
    for label in REVISIONS:
        if file_hash(opts.binaries / label) != reference[label]:
            raise RuntimeError('Binary differs from the prior comparison')
    lock_path = Path.home() / '.cache/fold-db-perf-guard/host-cargo.lock'
    index, fixtures = [], {}
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    with lock_path.open('a+') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX)
        for config in ('plain', 'preload'):
            args = argparse.Namespace(binary=opts.binaries / 'before',
                scratch=opts.scratch / f'fixture-{config}', output=opts.output / f'fixture-{config}',
                source_revision=REVISIONS['before'], config=config, rustc='rustc 1.96.1',
                client_cpu=opts.client_cpu, server_cpu=opts.server_cpu)
            fixtures[config] = execute(args)
            print(f'Fixture ready: {config}', flush=True)
        for pair in range(opts.rounds):
            for config in ('plain', 'preload'):
                order = ('before', 'after') if pair % 2 == 0 else ('after', 'before')
                for label in order:
                    name = f'{config}-{pair}-{label}'
                    args = argparse.Namespace(binary=opts.binaries / label,
                        scratch=opts.scratch / name, output=opts.output / name,
                        source_revision=REVISIONS[label], config=config, rustc='rustc 1.96.1',
                        client_cpu=opts.client_cpu, server_cpu=opts.server_cpu)
                    print(f'Starting {name}', flush=True)
                    fixture, summary = fixtures[config]
                    started = time.time()
                    bench, result = execute(args, fixture.home, summary['fixture_digest'])
                    cleanup(bench.root, bench.output)
                    index.append(dict(name=name, pair=pair, label=label, config=config,
                                      source_revision=REVISIONS[label], seconds=time.time() - started))
                    save(opts.output / 'index.json', index)
        for fixture, summary in fixtures.values():
            assert fixture_digest(fixture.home) == summary['fixture_digest']
            cleanup(fixture.root, fixture.output)
    print('All controlled measurements complete', flush=True)


if __name__ == '__main__':
    main()
